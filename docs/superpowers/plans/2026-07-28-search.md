> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a live search page at `/search` and clean up the landing page Hero by replacing the non-functional search bar with a scroll-to-demo button.

**Architecture:** A new `useSearch` composable holds all search state and Supabase queries. The `search.vue` page is a thin shell: search input on top, results list or `MultimodalTool` below depending on state. Hero.vue loses its search bar and gains a single CTA scroll button.

**Tech Stack:** Nuxt 4, Vue 3 Composition API, Supabase JS client (`useSupabaseClient`), Tailwind CSS v4, `MultimodalTool` component (already exists).

**Note:** No git commits — user handles all version control.

## Task 3: Create `useSearch` composable

**Files:**
- Create: `app/composables/useSearch.ts`

- [ ] **Step 1: Create the file with full implementation**

```ts
import { ref, watch } from 'vue'

export interface SearchResult {
  unNumber: string
  name: string | null
  nameEn: string | null
  nameFr: string | null
  hazardClass: string | null
  packingGroup: string | null
}

const UN_NUMBER_RE = /^\d{4}$/

export function useSearch() {
  const query = ref('')
  const results = ref<SearchResult[]>([])
  const selectedUn = ref<string | null>(null)
  const isLoading = ref(false)

  const supabase = useSupabaseClient()

  let debounceTimer: ReturnType<typeof setTimeout> | null = null

  async function search(q: string): Promise<void> {
    const trimmed = q.trim()
    if (trimmed.length < 4) {
      results.value = []
      return
    }

    isLoading.value = true
    try {
      const { data, error } = await supabase
        .from('adr_entries')
        .select('un_number, name, name_en, name_fr, hazard_class, packing_group')
        .or(`name.ilike.%${trimmed}%,name_en.ilike.%${trimmed}%,name_fr.ilike.%${trimmed}%`)
        .order('un_number')
        .limit(20)

      if (error) throw error
      results.value = (data ?? []).map(r => ({
        unNumber: String(r.un_number),
        name: r.name as string | null,
        nameEn: r.name_en as string | null,
        nameFr: r.name_fr as string | null,
        hazardClass: r.hazard_class as string | null,
        packingGroup: r.packing_group as string | null,
      }))
    } catch {
      results.value = []
    } finally {
      isLoading.value = false
    }
  }

  watch(query, (val) => {
    const trimmed = val.trim()
    if (debounceTimer) clearTimeout(debounceTimer)

    // 4-digit UN number: load tool immediately, skip debounce
    if (UN_NUMBER_RE.test(trimmed)) {
      results.value = []
      selectedUn.value = trimmed
      return
    }

    selectedUn.value = null
    debounceTimer = setTimeout(() => search(val), 300)
  })

  function select(unNumber: string) {
    selectedUn.value = unNumber
    results.value = []
  }

  return {
    query,
    results,
    selectedUn,
    isLoading,
    select,
  }
}
```
