Query pattern — UN number lookup
SELECT mode, hazard_class, packing_group,
       kemler_number, ems_fire, stowage_category
FROM un_comparison
WHERE un_number = '1203';

-- Returns 1 row per available mode
-- ADR · RID · IMDG · ICAO · UN
SELECT code, text_de
FROM special_provisions
WHERE mode = 'ADR'
  AND code = ANY('{"163","274","601"}');

-- Array containment — no JOIN needed
